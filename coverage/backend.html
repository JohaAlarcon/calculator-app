// Command server runs the calculator HTTP service.
//
// Configuration (environment):
//
//        PORT             listen port (default 8080)
//        ALLOWED_ORIGINS  comma-separated browser origins allowed via CORS (default http://localhost:3000)
package main

import (
        "context"
        "errors"
        "fmt"
        "log/slog"
        "net"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/JohaAlarcon/calculator-app/backend/internal/api"
)

// shutdownTimeout bounds how long in-flight requests may take to finish on SIGINT/SIGTERM.
const shutdownTimeout = 10 * time.Second

func main() {
        logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
        slog.SetDefault(logger)

        if err := runUntilSignalled(configFromEnv(os.Getenv), logger, nil); err != nil {
                logger.Error("server stopped with error", "error", err)
                os.Exit(1)
        }
}

// runUntilSignalled runs the service until the process receives SIGINT or SIGTERM, then
// shuts down gracefully (see run).
func runUntilSignalled(cfg serverConfig, logger *slog.Logger, onListening func(net.Addr)) error {
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        return run(ctx, cfg, logger, onListening)
}

// serverConfig is everything the service reads from its environment.
type serverConfig struct {
        Addr           string
        AllowedOrigins []string
}

// configFromEnv reads PORT and ALLOWED_ORIGINS, applying their defaults.
func configFromEnv(getenv func(string) string) serverConfig {
        port := getenv("PORT")
        if port == "" {
                port = "8080"
        }
        return serverConfig{Addr: ":" + port, AllowedOrigins: api.ParseAllowedOrigins(getenv("ALLOWED_ORIGINS"))}
}

// run serves the API on cfg.Addr until ctx is done, then shuts down gracefully, letting
// in-flight requests finish (up to shutdownTimeout). onListening, if set, receives the
// address actually listened on (useful with port 0).
func run(ctx context.Context, cfg serverConfig, logger *slog.Logger, onListening func(net.Addr)) error {
        listener, err := net.Listen("tcp", cfg.Addr)
        if err != nil {
                return fmt.Errorf("listen on %q: %w", cfg.Addr, err)
        }
        srv := &http.Server{
                Handler:           api.NewRouter(api.Config{AllowedOrigins: cfg.AllowedOrigins, Logger: logger}),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
                ErrorLog:          slog.NewLogLogger(logger.Handler(), slog.LevelError),
        }

        serveErr := make(chan error, 1)
        go func() { serveErr <- srv.Serve(listener) }()
        logger.Info("calculator service listening", "addr", listener.Addr().String(), "allowed_origins", cfg.AllowedOrigins)
        if onListening != nil {
                onListening(listener.Addr())
        }

        select {
        case err := <-serveErr:
                return err
        case <-ctx.Done():
        }

        logger.Info("shutting down: draining in-flight requests", "timeout", shutdownTimeout.String())
        shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return err
        }
        if err := <-serveErr; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        logger.Info("shutdown complete")
        return nil
}

// Package api is the HTTP adapter over the calculator domain.
package api

import (
        "bytes"
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"

        "github.com/JohaAlarcon/calculator-app/backend/internal/calculator"
)

type calculateRequest struct {
        Operation string    `json:"operation"`
        Operands  []float64 `json:"operands"`
}

type calculateResponse struct {
        calculateRequest
        Result float64 `json:"result"`
}

// NewRouter returns the service's HTTP handler: all routes, wrapped in ServiceMiddleware.
func NewRouter(cfg Config) http.Handler {
        mux := http.NewServeMux()
        route(mux, http.MethodPost, "/api/v1/calculate", handleCalculate)
        route(mux, http.MethodGet, "/api/v1/operations", handleListOperators)
        route(mux, http.MethodGet, "/healthz", handleHealthz)
        mux.HandleFunc("/", func(w http.ResponseWriter, _ *http.Request) {
                writeError(w, http.StatusNotFound, codeNotFound, "No such endpoint")
        })
        return ServiceMiddleware(cfg)(mux)
}

// route registers a handler for one method on a path, and a JSON 405 (with an
// Allow header) for every other method on that path — the standard library's
// own 405 and 404 replies are plain text, but every error body must be JSON.
func route(mux *http.ServeMux, method, path string, handler http.HandlerFunc) {
        mux.HandleFunc(method+" "+path, handler)
        mux.HandleFunc(path, func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Allow", method)
                writeError(w, http.StatusMethodNotAllowed, codeMethodNotAllowed,
                        fmt.Sprintf("Method %s is not allowed on %s; use %s", r.Method, path, method))
        })
}

// maxRequestBodyBytes bounds the calculate request body (1 KB).
const maxRequestBodyBytes = 1024

func handleCalculate(w http.ResponseWriter, r *http.Request) {
        req, err := decodeCalculateRequest(w, r)
        if err != nil {
                writeError(w, http.StatusBadRequest, codeInvalidJSON, err.Error())
                return
        }
        result, err := calculator.Evaluate(req.Operation, req.Operands)
        if err != nil {
                writeOperationError(w, err)
                return
        }
        writeJSON(w, http.StatusOK, calculateResponse{calculateRequest: req, Result: result})
}

// decodeCalculateRequest reads exactly one JSON object with only the known fields,
// within the body-size limit. Its errors are user-facing messages.
func decodeCalculateRequest(w http.ResponseWriter, r *http.Request) (calculateRequest, error) {
        var req calculateRequest
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxRequestBodyBytes))
        dec.DisallowUnknownFields()
        err := dec.Decode(&req)
        if err == nil {
                // Exactly one object: the next read must hit the end of the body.
                switch next := dec.Decode(&struct{}{}); {
                case next == nil:
                        err = errors.New("trailing data after the JSON object")
                case next != io.EOF:
                        err = next // malformed trailing data, or the body-size limit
                }
        }
        if err == nil {
                return req, nil
        }
        var tooLarge *http.MaxBytesError
        if errors.As(err, &tooLarge) {
                return req, fmt.Errorf("Request body must not exceed %d KB", maxRequestBodyBytes/1024)
        }
        return req, errors.New(`Request body must be one JSON object like {"operation":"add","operands":[2,3]}`)
}

// operatorsResponse is the body of GET /api/v1/operations; the path and the
// "operations" key are fixed by the API contract (spec #1).
type operatorsResponse struct {
        Operators []calculator.OperatorInfo `json:"operations"`
}

func handleListOperators(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, operatorsResponse{Operators: calculator.Operators()})
}

func handleHealthz(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

// writeJSON encodes before writing the status, so an unencodable body
// (e.g. a non-finite float) becomes a 500 instead of a 200 with an empty body.
func writeJSON(w http.ResponseWriter, status int, body any) {
        var buf bytes.Buffer
        if err := json.NewEncoder(&buf).Encode(body); err != nil {
                slog.Error("encode response", "error", err)
                writeInternalError(w) // re-enters writeJSON with a body that always encodes
                return
        }
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _, _ = w.Write(buf.Bytes())
}

// errorCode is a stable, machine-readable error identifier; part of the public API contract (ADR-0004).
type errorCode string

const (
        codeInternalError    errorCode = "INTERNAL_ERROR"
        codeNotFound         errorCode = "NOT_FOUND"
        codeMethodNotAllowed errorCode = "METHOD_NOT_ALLOWED"

        // Invalid Requests (HTTP 400).
        codeInvalidJSON         errorCode = "INVALID_JSON"
        codeUnknownOperation    errorCode = "UNKNOWN_OPERATION"
        codeInvalidOperandCount errorCode = "INVALID_OPERAND_COUNT"

        // Math Errors (HTTP 422).
        codeDivisionByZero errorCode = "DIVISION_BY_ZERO"
        codeInvalidOperand errorCode = "INVALID_OPERAND"
        codeOverflow       errorCode = "OVERFLOW"
)

// errorResponse is the JSON shape of every error (spec #1): {"error":{"code","message"}}.
type errorResponse struct {
        Error errorBody `json:"error"`
}

type errorBody struct {
        Code    errorCode `json:"code"`
        Message string    `json:"message"`
}

func writeError(w http.ResponseWriter, status int, code errorCode, message string) {
        writeJSON(w, status, errorResponse{Error: errorBody{Code: code, Message: message}})
}

func writeInternalError(w http.ResponseWriter) {
        writeError(w, http.StatusInternalServerError, codeInternalError, "Internal server error")
}

// mathErrors maps each domain Math Error to its public code and message (ADR-0004).
var mathErrors = []struct {
        err     error
        code    errorCode
        message string
}{
        {calculator.ErrDivisionByZero, codeDivisionByZero, "Cannot divide by zero"},
        {calculator.ErrInvalidOperand, codeInvalidOperand, "Result is not a real number"},
        {calculator.ErrOverflow, codeOverflow, "Result is too large"},
}

// writeOperationError answers a failed Evaluate: an Invalid Request is 400,
// a Math Error is 422 (ADR-0004).
func writeOperationError(w http.ResponseWriter, err error) {
        var unknown *calculator.UnknownOperatorError
        if errors.As(err, &unknown) {
                writeError(w, http.StatusBadRequest, codeUnknownOperation, fmt.Sprintf("Unknown operator %q", unknown.Operator))
                return
        }
        var count *calculator.OperandCountError
        if errors.As(err, &count) {
                writeError(w, http.StatusBadRequest, codeInvalidOperandCount,
                        fmt.Sprintf("Operator %q takes %d operands, got %d", count.Operator, count.Want, count.Got))
                return
        }
        for _, m := range mathErrors {
                if errors.Is(err, m.err) {
                        writeError(w, http.StatusUnprocessableEntity, m.code, m.message)
                        return
                }
        }
        slog.Error("unclassified Operation error", "error", err)
        writeInternalError(w)
}

package api

import (
        "log/slog"
        "net/http"
        "slices"
        "strings"
        "time"
)

// DefaultAllowedOrigin is the UI's origin when ALLOWED_ORIGINS is not set.
const DefaultAllowedOrigin = "http://localhost:3000"

// Config holds the HTTP layer's settings.
type Config struct {
        // AllowedOrigins are the exact browser origins allowed to call the service
        // directly (ADR-0005). "*" is never treated as a wildcard.
        AllowedOrigins []string
        // Logger receives one structured entry per request; nil means slog.Default().
        Logger *slog.Logger
}

// ParseAllowedOrigins turns a comma-separated ALLOWED_ORIGINS value into a list,
// trimming blanks; an empty value yields DefaultAllowedOrigin.
func ParseAllowedOrigins(raw string) []string {
        var origins []string
        for _, origin := range strings.Split(raw, ",") {
                if origin = strings.TrimSpace(origin); origin != "" {
                        origins = append(origins, origin)
                }
        }
        if len(origins) == 0 {
                return []string{DefaultAllowedOrigin}
        }
        return origins
}

// ServiceMiddleware wraps a handler with the service's cross-cutting behaviour:
// request logging, panic recovery and CORS (outermost first).
func ServiceMiddleware(cfg Config) func(http.Handler) http.Handler {
        logger := cfg.Logger
        if logger == nil {
                logger = slog.Default()
        }
        return func(next http.Handler) http.Handler {
                return logRequests(logger, recoverPanics(logger, cors(cfg.AllowedOrigins, next)))
        }
}

// cors answers preflight requests and adds CORS headers for allowed origins only.
func cors(allowedOrigins []string, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                origin := r.Header.Get("Origin")
                allowed := origin != "" && slices.Contains(allowedOrigins, origin)
                if origin != "" {
                        w.Header().Add("Vary", "Origin")
                }
                if allowed {
                        w.Header().Set("Access-Control-Allow-Origin", origin)
                }

                isPreflight := r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != ""
                if isPreflight {
                        if !allowed {
                                w.WriteHeader(http.StatusForbidden)
                                return
                        }
                        // GET is a CORS-safelisted method and never needs a preflight, so only POST is listed.
                        w.Header().Set("Access-Control-Allow-Methods", "POST, OPTIONS")
                        w.Header().Set("Access-Control-Allow-Headers", "Content-Type")
                        w.Header().Set("Access-Control-Max-Age", "600")
                        w.WriteHeader(http.StatusNoContent)
                        return
                }
                next.ServeHTTP(w, r)
        })
}

// recoverPanics turns a panicking handler into a 500 JSON error and keeps the service up.
// The panic value is logged, never sent to the client. If the handler had already started
// the response, it is too late for a clean error, so the connection is aborted instead.
func recoverPanics(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                defer func() {
                        v := recover()
                        if v == nil {
                                return
                        }
                        if v == http.ErrAbortHandler {
                                panic(v)
                        }
                        logger.Error("panic serving request", "method", r.Method, "path", r.URL.Path, "panic", v)
                        if rec.wroteHeader {
                                panic(http.ErrAbortHandler)
                        }
                        writeInternalError(w)
                }()
                next.ServeHTTP(rec, r)
        })
}

// logRequests writes one structured log entry per request once it completes.
func logRequests(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                next.ServeHTTP(rec, r)
                logger.Info("request",
                        "method", r.Method,
                        "path", r.URL.Path,
                        "status", rec.status,
                        "duration_ms", float64(time.Since(start).Microseconds())/1000,
                )
        })
}

// statusRecorder remembers the status code a handler wrote and whether it started the response.
type statusRecorder struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (r *statusRecorder) WriteHeader(status int) {
        if !r.wroteHeader {
                r.status = status
                r.wroteHeader = true
        }
        r.ResponseWriter.WriteHeader(status)
}

func (r *statusRecorder) Write(b []byte) (int, error) {
        r.wroteHeader = true
        return r.ResponseWriter.Write(b)
}

// Unwrap lets http.ResponseController reach the underlying writer.
func (r *statusRecorder) Unwrap() http.ResponseWriter {
        return r.ResponseWriter
}

// Package calculator is the arithmetic domain: it evaluates a single Operation
// (an Operator applied to its Operands) and knows nothing about HTTP.
package calculator

import (
        "cmp"
        "errors"
        "fmt"
        "math"
        "slices"
)

// UnknownOperatorError is returned when no Operator with the given name exists.
type UnknownOperatorError struct {
        Operator string
}

func (e *UnknownOperatorError) Error() string { return fmt.Sprintf("unknown operator %q", e.Operator) }

// OperandCountError is returned when the number of Operands does not match the Operator's arity.
type OperandCountError struct {
        Operator  string
        Want, Got int
}

func (e *OperandCountError) Error() string {
        return fmt.Sprintf("operator %q takes %d operands, got %d", e.Operator, e.Want, e.Got)
}

// Math Errors: a well-formed Operation with no finite real Result (ADR-0004).
var (
        // ErrDivisionByZero is returned when an Operation would divide by zero.
        ErrDivisionByZero = errors.New("division by zero")
        // ErrInvalidOperand is returned when an Operand is outside the Operator's domain
        // (the Result would not be a real number).
        ErrInvalidOperand = errors.New("operand outside the operator's domain")
        // ErrOverflow is returned when the Result is too large to represent.
        ErrOverflow = errors.New("result too large")
)

// operator is a registry entry: how many Operands it takes, an optional domain
// check that reports a Math Error before evaluating, and how to evaluate them.
type operator struct {
        arity int
        check func(operands []float64) error
        eval  func(operands []float64) float64
}

// registry holds every supported Operator. Adding an Operator is one entry here.
var registry = map[string]operator{
        "add":      {arity: 2, eval: func(o []float64) float64 { return o[0] + o[1] }},
        "subtract": {arity: 2, eval: func(o []float64) float64 { return o[0] - o[1] }},
        "multiply": {arity: 2, eval: func(o []float64) float64 { return o[0] * o[1] }},
        "divide":   {arity: 2, check: nonZeroDivisor, eval: func(o []float64) float64 { return o[0] / o[1] }},
        // percentage(a, b) is "a percent of b". It is context-free: the app picks b
        // from the Chain (ADR-0002), e.g. percentage(10, 1) for a plain ÷100.
        // Dividing first avoids a spurious overflow in the a×b product.
        "percentage": {arity: 2, eval: func(o []float64) float64 { return o[0] / 100 * o[1] }},
        // A negative base with a fractional exponent yields NaN → ErrInvalidOperand (via Evaluate).
        "power": {arity: 2, check: nonZeroBaseForNegativeExponent, eval: func(o []float64) float64 { return math.Pow(o[0], o[1]) }},
        // √ of a negative yields NaN → ErrInvalidOperand (via Evaluate).
        "sqrt": {arity: 1, eval: func(o []float64) float64 { return math.Sqrt(o[0]) }},
}

func nonZeroDivisor(o []float64) error {
        if o[1] == 0 {
                return ErrDivisionByZero
        }
        return nil
}

// nonZeroBaseForNegativeExponent: 0 raised to a negative power is 1 ÷ 0ⁿ, a division by zero
// (math.Pow would return +Inf, which would otherwise be reported as an overflow).
func nonZeroBaseForNegativeExponent(o []float64) error {
        if o[0] == 0 && o[1] < 0 {
                return ErrDivisionByZero
        }
        return nil
}

// Evaluate applies the named Operator to the Operands and returns the Result.
// It never returns a non-finite Result: NaN becomes ErrInvalidOperand and
// ±Inf becomes ErrOverflow, unless the Operator's own check reported a more
// specific Math Error first.
func Evaluate(name string, operands []float64) (float64, error) {
        op, ok := registry[name]
        if !ok {
                return 0, &UnknownOperatorError{Operator: name}
        }
        if len(operands) != op.arity {
                return 0, &OperandCountError{Operator: name, Want: op.arity, Got: len(operands)}
        }
        if op.check != nil {
                if err := op.check(operands); err != nil {
                        return 0, err
                }
        }
        result := op.eval(operands)
        switch {
        case math.IsNaN(result):
                return 0, ErrInvalidOperand
        case math.IsInf(result, 0):
                return 0, ErrOverflow
        }
        return result, nil
}

// OperatorInfo describes a supported Operator as advertised to clients: its name and
// how many Operands it takes. Name is a plain string because clients may receive
// Operators they don't know yet (the frontend's drift check relies on that).
type OperatorInfo struct {
        Name  string `json:"name"`
        Arity int    `json:"arity"`
}

// Operators lists every supported Operator, sorted by name, straight from the registry.
func Operators() []OperatorInfo {
        infos := make([]OperatorInfo, 0, len(registry))
        for name, entry := range registry {
                infos = append(infos, OperatorInfo{Name: name, Arity: entry.arity})
        }
        slices.SortFunc(infos, func(a, b OperatorInfo) int { return cmp.Compare(a.Name, b.Name) })
        return infos
}
